function generate() { const body = document.getElementById('combiBody'); body.innerHTML = ''; let validCount = 1; // [수정] 전체 풀을 낮은 수(Low)와 높은 수(High)로 분리 const pool = [...new Set([...missingNums, ...hotNums])]; const lowPool = pool.filter(n => n <= 22); const highPool = pool.filter(n => n > 22); while (validCount <= 10) { let res = new Set(); // [핵심] 합계 분산을 위해 저번호 3개 + 고번호 3개를 각각 추출 // 만약 특정 풀의 숫자가 부족할 경우를 대비해 유연하게 설계 while(res.size < 3 && lowPool.length >= 3) { res.add(lowPool[Math.floor(Math.random() * lowPool.length)]); } while(res.size < 6) { res.add(highPool[Math.floor(Math.random() * highPool.length)]); } const sorted = Array.from(res).sort((a,b) => a-b); // 필터 1: 합계 범위를 100 ~ 190으로 유연하게 조정 (쏠림 방지) const sum = sorted.reduce((a, b) => a + b, 0); if (sum < 100 || sum > 190) continue; // 필터 2: 홀짝 균형 (1:5 ~ 5:1 비율) const odds = sorted.filter(n => n % 2 !== 0).length; if (odds === 0 || odds === 6) continue; // 필터 3: AC 값 및 3연번 금지 if (getAC(sorted) < 6) continue; let triple = false; for(let k=0; k<=3; k++) if(sorted[k]+1===sorted[k+1] && sorted[k+1]+1===sorted[k+2]) triple=true; if(triple) continue; const tr = document.createElement('tr'); const ballsHtml = sorted.map(n => { let cls = missingNums.includes(n) ? 'missing' : (hotNums.includes(n) ? 'hot' : ''); return `
${n}
`; }).join(''); tr.innerHTML = `${validCount}
${ballsHtml}
합계: ${sum}
`; body.appendChild(tr); validCount++; } }
